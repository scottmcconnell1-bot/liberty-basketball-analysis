        passes[:] = [
            p for p in passes
            if not (
                (not p.get("orphan"))
                and {p.get("fromPid"), p.get("toPid")} == {"o2", "o3"}
            )
        ]
        return


def apply_pitt5_sequence_routes(
    crop_gray,
    digit_pos: dict,
    paths: dict,
    marks: dict,
    passes: list,
    image_path: str | Path | None = None,
) -> None:
    """Scott's Pitt 5 sequence when OCR/ink order is wrong.

    Opening (page 172 ink, or opening formation): Pass 1→5, Cut 1→left corner,
    Relocate 2→right corner (same phase as 1's cut).
    Screen (page 173): 4 screens for 5, 5 cuts to right block, 4 drops to left block.
    """
    if not digit_pos:
        return

    stem = Path(image_path).stem.lower() if image_path else ""
    role = None
    if "0172" in stem or _pitt5_opening_formation(digit_pos):
        role = "opening"
    elif "0173" in stem or _pitt5_screen_formation(digit_pos):
        role = "screen"
    else:
        return

    def _ensure_pass(frm: str, to: str) -> None:
        nonlocal passes
        if any(
            (not p.get("orphan")) and p.get("fromPid") == frm and p.get("toPid") == to
            for p in passes
        ):
            return
        a = digit_pos.get(frm)
        b = digit_pos.get(to)
        if not _has_xy(a) or not _has_xy(b):
            return
        poly = trace_ink_polyline(crop_gray, a, b, digit_positions=digit_pos)
        if len(poly) < 2:
            poly = [
                {"x": float(a["x"]), "y": float(a["y"])},
                {"x": float(b["x"]), "y": float(b["y"])},
            ]
        passes[:] = [
            p
            for p in passes
            if p.get("orphan")
            or not (
                {p.get("fromPid"), p.get("toPid")} == {frm, to}
                or (p.get("fromPid") == frm and p.get("toPid") != to)
            )
        ]
        # Drop orphan duplicates of this pass corridor.
        passes[:] = [
            p
            for p in passes
            if not (
                p.get("orphan")
                and {p.get("fromPid"), p.get("toPid")} == {frm, to}
            )
        ]
        passes.append({
            "fromPid": frm,
            "toPid": to,
            "points": poly,
            "type": "pass",
        })

    def _ensure_route(oid: str, dest: dict, kind: str, *, min_disp: float = 40.0) -> None:
        start = digit_pos.get(oid)
        if not _has_xy(start) or not _has_xy(dest):
            return
        disp = (
            (float(dest["x"]) - float(start["x"])) ** 2
            + (float(dest["y"]) - float(start["y"])) ** 2
        ) ** 0.5
        if disp < min_disp:
            return
        poly = trace_ink_polyline(crop_gray, start, dest, digit_positions=digit_pos)
        if len(poly) < 2 or _mid_raw_ink_fraction(crop_gray, poly, start, dest) < 0.15:
            poly = [
                {"x": float(start["x"]), "y": float(start["y"])},
                {"x": float(dest["x"]), "y": float(dest["y"])},
            ]
        else:
            poly = [dict(pt) for pt in poly]
            poly[0] = {"x": float(start["x"]), "y": float(start["y"])}
            poly[-1] = {"x": float(dest["x"]), "y": float(dest["y"])}
        paths[oid] = poly
        marks[oid] = kind

    if role == "opening":
        o1, o2, o4, o5 = (digit_pos.get(f"o{i}") for i in (1, 2, 4, 5))
        _ensure_pass("o1", "o5")
        left_corner = {"x": 22.0, "y": 78.0}
        right_corner = {"x": 478.0, "y": 76.0}
        # Prefer sheet ink tips when present.
        if _has_xy(o1):
            ink_1 = trace_ink_polyline(
                crop_gray, o1, left_corner, digit_positions=digit_pos
            )
            if len(ink_1) >= 2 and _mid_raw_ink_fraction(crop_gray, ink_1, o1, left_corner) >= 0.15:
                tip = ink_1[-1]
                if float(tip["x"]) < 120 and float(tip["y"]) < 160:
                    left_corner = {"x": float(tip["x"]), "y": float(tip["y"])}
        if _has_xy(o2):
            ink_2 = trace_ink_polyline(
                crop_gray, o2, right_corner, digit_positions=digit_pos
            )
            if len(ink_2) >= 2 and _mid_raw_ink_fraction(crop_gray, ink_2, o2, right_corner) >= 0.15:
                tip = ink_2[-1]
                if float(tip["x"]) > 380 and float(tip["y"]) < 160:
                    right_corner = {"x": float(tip["x"]), "y": float(tip["y"])}
        _ensure_route("o1", left_corner, "cut", min_disp=50.0)
        _ensure_route("o2", right_corner, "cut", min_disp=40.0)
        # Drop false receiver walk along the pass and idle bigs.
        for oid in ("o3", "o4", "o5"):
            if marks.get(oid) in ("cut", "dribble", "screen"):
                paths.pop(oid, None)
                marks.pop(oid, None)
        return

    # Screen sheet: 4 up to screen 5, 5 to right block, 4 to left block.
    o4, o5 = digit_pos.get("o4"), digit_pos.get("o5")
    if not _has_xy(o4) or not _has_xy(o5):
        return
    screen_spot = {
        "x": float(o5["x"]) + 18.0,
        "y": max(160.0, float(o5["y"]) - 35.0),
    }
    ink_screen = trace_ink_polyline(crop_gray, o4, o5, digit_positions=digit_pos)
    if len(ink_screen) >= 2 and _mid_raw_ink_fraction(crop_gray, ink_screen, o4, o5) >= 0.12:
        # T-bar tip is short of 5's glyph — use ~85% along the stem.
        tip = ink_screen[min(len(ink_screen) - 1, max(2, int(len(ink_screen) * 0.85)))]
        screen_spot = {"x": float(tip["x"]), "y": float(tip["y"])}
    _ensure_route("o4", screen_spot, "screen", min_disp=35.0)

    right_block = {"x": float(o4["x"]), "y": float(o4["y"])}
    left_block = {
        "x": max(120.0, min(200.0, 500.0 - float(o4["x"]))),
        "y": float(o4["y"]),
    }
    # 5's cut toward the paint / right block — follow ink when strong.
    basketish = {"x": 250.0, "y": 55.0}
    ink_5 = trace_ink_polyline(crop_gray, o5, right_block, digit_positions=digit_pos)
    if len(ink_5) < 2 or _mid_raw_ink_fraction(crop_gray, ink_5, o5, right_block) < 0.12:
        ink_5 = trace_ink_polyline(crop_gray, o5, basketish, digit_positions=digit_pos)
    if len(ink_5) >= 2 and _mid_raw_ink_fraction(
        crop_gray, ink_5, o5, ink_5[-1]
    ) >= 0.12:
        tip = ink_5[-1]
        # Prefer right-side paint finish (right block), not left.
        if float(tip["x"]) < 220:
            tip = {"x": float(right_block["x"]), "y": min(140.0, float(tip["y"]))}
        right_block = {"x": float(tip["x"]), "y": float(tip["y"])}
    _ensure_route("o5", right_block, "cut", min_disp=40.0)

    # After screening, 4 drops to the left block (roll / replace).
    ink_4drop = trace_ink_polyline(crop_gray, screen_spot, left_block, digit_positions=digit_pos)
    if len(ink_4drop) >= 2 and _mid_raw_ink_fraction(
        crop_gray, ink_4drop, screen_spot, left_block
    ) >= 0.12:
        tip = ink_4drop[-1]
        if float(tip["x"]) < 260:
            left_block = {"x": float(tip["x"]), "y": float(tip["y"])}
    # Second move for 4: store drop tip on a synthetic continue by overwriting
    # only after screen land — animator uses heldLandings; keep drop as o4 cut
    # destination from screen_spot by encoding full path screen→left in points
    # when screen already set. Prefer explicit drop route from current o4 if
    # screen tip ≈ o4 (rare); else path from screen_spot.
    start_4 = screen_spot if marks.get("o4") == "screen" else o4
    drop_poly = trace_ink_polyline(crop_gray, start_4, left_block, digit_positions=digit_pos)
    if len(drop_poly) < 2 or _mid_raw_ink_fraction(
        crop_gray, drop_poly, start_4, left_block
    ) < 0.12:
        drop_poly = [
            {"x": float(start_4["x"]), "y": float(start_4["y"])},
            {"x": float(left_block["x"]), "y": float(left_block["y"])},
        ]
    else:
        drop_poly = [dict(pt) for pt in drop_poly]
        drop_poly[0] = {"x": float(start_4["x"]), "y": float(start_4["y"])}
        drop_poly[-1] = {"x": float(left_block["x"]), "y": float(left_block["y"])}
    # Keep screen route on o4; stash drop as a second path via marks note.
    # Frontend Pitt 5 ordering emits Screen o4 then Relocate o4 from held tip.
    # Encode drop destination on a dedicated key consumed by the animator.
    paths["o4_drop"] = drop_poly
    marks["o4_drop"] = "cut"

    # No passes / guard cuts on the screen sheet.
    for oid in ("o1", "o2", "o3"):
        paths.pop(oid, None)
        marks.pop(oid, None)
    passes[:] = [p for p in passes if p.get("orphan")]
    passes.clear()
