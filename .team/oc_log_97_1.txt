+function Write-RoundStartSnapshot {
+    $map = [ordered]@{}
+    try {
+        $statusLines = @(& $GitExe status --porcelain 2>$null)
+    } catch {
+        $statusLines = @()
+    }
+    foreach ($line in $statusLines) {
+        if ($null -eq $line) {
+            continue
+        }
+        $text = [string]$line
+        if ($text.Length -lt 4) {
+            continue
+        }
+        $path = $text.Substring(3)
+        if ($path -match ' -> ') {
+            $path = ($path -split ' -> ', 2)[1]
+        }
+        $path = $path.Trim()
+        if ($path.Length -ge 2 -and $path.StartsWith('"') -and $path.EndsWith('"')) {
+            $path = $path.Substring(1, $path.Length - 2)
+        }
+        $path = $path -replace '\\', '/'
+        while ($path.StartsWith('./')) {
+            $path = $path.Substring(2)
+        }
+        if (-not $path) {
+            continue
+        }
+        $full = Join-Path $Repo $path
+        $hash = 'deleted'
+        if (Test-Path -LiteralPath $full -PathType Leaf) {
+            try {
+                $hash = (Get-FileHash -LiteralPath $full -Algorithm SHA256).Hash.ToLowerInvariant()
+            } catch {
+                $hash = 'deleted'
+            }
+        }
+        $map[$path] = $hash
+    }
+    $json = $map | ConvertTo-Json -Compress
+    if (-not $json) {
+        $json = '{}'
+    }
+    Write-TextFile -Path $RoundStartPath -Text ($json + $NewLine)
+}
