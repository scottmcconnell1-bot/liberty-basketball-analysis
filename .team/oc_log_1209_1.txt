import json
import numpy as np
import cv2
from pathlib import Path

ROOT = Path(r'C:\Users\scott\Documents\liberty-basketball-analysis')
ROI_FILE = ROOT / "data" / "roi" / "notus_scoreboard_rois.json"
EYE_READ_FILE = ROOT / ".team" / "notus_fit_crops" / "eye_read_values.json"
GEOMETRY_FILE = ROOT / "data" / "roi" / "notus_segment_geometry.json"

def load_json(p): return json.loads(p.read_text(encoding="utf-8"))

rois = load_json(ROI_FILE)
eye_reads = load_json(EYE_READ_FILE)
geometry = load_json(GEOMETRY_FILE)["segments"]

def get_on_segments(digit):
    # Patterns based on _SEVEN_SEGMENT from scoreboard_clock.py logic
    # 0: abcefg, 1: bc, 2: abdeg, 3: abcdg, 4: bcfg, 5: acdfg, 6: acdefg, 7: abc, 8: abcdefg, 9: abcfg
    patterns = {
        0: "abcefg", 1: "bc", 2: "abdeg", 3: "abcdg", 4: "bcfg",
        5: "acdfg", 6: "acdefg", 7: "abc", 8: "abcdefg", 9: "abcfg"
    }
    return patterns.get(digit, "")

def measure_game(game_key):
    game_id = [k for k, v in rois["games"].items() if game_key in k][0]
    video_path = ROOT / rois["games"][game_id]["video"]
    
    pooled_lit_pixels = []
    
    # Find entries for this game
    entries = [e for e in eye_reads["entries"] if e["game"] == game_key]
    
    cap = cv2.VideoCapture(str(video_path))
    
    for e in entries:
        digit = e["digit"]
        if digit is None: continue
        
        # Extract crop from video at timestamp
        # The crop was originally cut to the box. 
        # We need the box for the slot.
        slot = e["slot"]
        # Slot keys in ROI file: "home_tens", "home_ones", "away_tens", "away_ones"
        # Simplified: use the 'scores' map
        score_side = "home" if "home" in slot else "away"
        slot_type = "tens" if "tens" in slot else "ones"
        box = rois["games"][game_id]["scores"][score_side][slot_type]
        
        cap.set(cv2.CAP_PROP_POS_MSEC, e["timestamp_s"] * 1000)
        ret, frame = cap.read()
        if not ret: continue
        
        x, y, w, h = box
        crop = frame[y:y+h, x:x+w]
        hsv = cv2.cvtColor(crop, cv2.COLOR_BGR2HSV)
        
        # Determine ON segments for this digit
        on_segs_str = get_on_segments(digit)
        on_pixels_mask = np.zeros((h, w), dtype=bool)
        
        for seg in on_segs_str:
            fx0, fy0, fx1, fy1 = geometry[seg]
            px0, py0 = int(fx0 * w), int(fy0 * h)
            px1, py1 = int(fx1 * w), int(fy1 * h)
            on_pixels_mask[py0:py1, px0:px1] = True
        
        on_region_v = hsv[on_pixels_mask, 2]
        if on_region_v.size == 0: continue
        
        median_v = np.median(on_region_v)
        lit_mask = (hsv[:, :, 2] >= median_v) & on_pixels_mask
        pooled_lit_pixels.append(hsv[lit_mask])
        
    cap.release()
    
    if not pooled_lit_pixels:
        return None
        
    all_lit = np.vstack(pooled_lit_pixels)
    hues = all_lit[:, 0]
    sats = all_lit[:, 1]
    vals = all_lit[:, 2]
    
    # Hue handling: map >165 to hue-180
    hues_shifted = np.where(hues > 165, hues - 180, hues)
    h_min = np.percentile(hues_shifted, 1)
    h_max = np.percentile(hues_shifted, 99)
    
    # Map back
    if h_min < 0: h_min += 180
    if h_max < 0: h_max += 180
    
    # Actually, the plan says map hues > 165 to hue-180, compute percentiles, then map back.
    # If we map hues > 165 to hue-180, the red cluster (around 0) becomes contiguous (around 0).
    # Example: 175 -> -5, 5 -> 5. Cluster is [-5, 5].
    # Percentiles of [-5, 5] might be -4, 4.
    # Map back: -4 -> 176, 4 -> 4. Range [176, 180) U [0, 4].
    
    # Recalculate properly:
    h_shifted = np.where(hues > 165, hues - 180, hues)
    p1, p99 = np.percentile(h_shifted, [1, 99])
    
    final_h_min = p1 if p1 >= 0 else p1 + 180
    final_h_max = p99 if p99 >= 0 else p99 + 180
    
    # S and V 10th percentiles
    s_min = np.percentile(sats, 10)
    v_min = np.percentile(vals, 10)
    
    return {"h_min": float(final_h_min), "h_max": float(final_h_max), "s_min": float(s_min), "v_min": float(v_min)}

import json
import numpy as np
import cv2
from pathlib import Path

ROOT = Path(r'C:\Users\scott\Documents\liberty-basketball-analysis')
ROI_FILE = ROOT / "data" / "roi" / "notus_scoreboard_rois.json"
EYE_READ_FILE = ROOT / ".team" / "notus_fit_crops" / "eye_read_values.json"
GEOMETRY_FILE = ROOT / "data" / "roi" / "notus_segment_geometry.json"

def load_json(p): return json.loads(p.read_text(encoding="utf-8"))

rois = load_json(ROI_FILE)
eye_reads = load_json(EYE_READ_FILE)
geometry = load_json(GEOMETRY_FILE)["segments"]

def get_on_segments(digit):
    patterns = {
        0: "abcefg", 1: "bc", 2: "abdeg", 3: "abcdg", 4: "bcfg",

    }
    return patterns.get(digit, "")

def measure_game(game_key):

    video_path = ROOT / rois["games"][game_id]["video"]

            px0, py0 = int(fx0 * w), int(fy0 * h)

    return {"h_min": float(final_h_min), "h_max": float(final_h_max), "s_min": float(s_min), "v_min": float(v_min)}

import json

import cv2
from pathlib import Path

ROI_FILE = ROOT / "data" / "roi" / "notus_scoreboard_rois.json"

def get_on_segments(digit):

        0: "abcefg", 1: "bc", 2: "abdeg", 3: "abcdg", 4: "bcfg",

            crop = frame[y:y+h, x:x+w]
            hsv = cv2.cvtColor(crop, cv2.COLOR_BGR2HSV)
