function orderPitt5Beats(beats, sheetIndex) {
  if (!beats || !beats.length || !isPitt5Play()) return beats;
  const preferred = [];
  if (sheetIndex === 0) {
    preferred.push(
      (b) => b.kind === 'pass' && b.path && b.path.fromPid === 'o1' && b.path.toPid === 'o5',
      (b) => (b.kind === 'cut' || b.kind === 'relocate') && b.path && b.path.pid === 'o1',
      (b) => (b.kind === 'cut' || b.kind === 'relocate') && b.path && b.path.pid === 'o2',
    );
  } else {
    preferred.push(
      (b) => b.kind === 'screen' && b.path && b.path.pid === 'o4',
      (b) => (b.kind === 'cut' || b.kind === 'relocate') && b.path && b.path.pid === 'o5',
      (b) => (b.kind === 'cut' || b.kind === 'relocate') && b.path && b.path.pid === 'o4'
        && (b.path.pitt5Drop || b.path.type !== 'screen'),
    );
  }
  const remaining = beats.slice();
  const ordered = [];
  preferred.forEach((pred) => {
    for (let i = 0; i < remaining.length; i += 1) {
      if (pred(remaining[i])) {
        ordered.push(remaining.splice(i, 1)[0]);
        break;
      }
    }
  });
  return ordered.concat(remaining);
}

/** Fill missing Pitt 5 beats from sheet deltas / ink when OCR order was wrong. */
function ensurePitt5Beats(beats, fromRaw, toRaw, pathList, ballPid, sheetIndex) {
  if (!isPitt5Play()) return beats || [];
  const out = (beats || []).slice();
  const hasPass15 = out.some(
    (b) => b.kind === 'pass' && b.path && b.path.fromPid === 'o1' && b.path.toPid === 'o5',
  );
  const hasMover = (pid, kinds) => out.some(
    (b) => kinds.includes(b.kind) && b.path && b.path.pid === pid,
  );
  const idx = typeof sheetIndex === 'number' ? sheetIndex : 0;

  if (idx === 0) {
    if (!hasPass15) {
      const a = fromRaw.o1;
      const b = fromRaw.o5 || toRaw.o5;
      if (a && b) {
        out.unshift({
          kind: 'pass',
          label: 'Pass #1 → #5',
          path: {
            pid: null,
            ballOnly: true,
            from: { x: a.x, y: a.y },
            to: { x: b.x, y: b.y },
            ctrl: null,
            points: null,
            type: 'pass',
            fromPid: 'o1',
            toPid: 'o5',
          },
          holdAfterMs: ANIM.HOLD_PASS_MS,
          ballTo: 'o5',
        });
      }
    }
    ['o1', 'o2'].forEach((pid) => {
      if (hasMover(pid, ['cut', 'relocate'])) return;
      const a = fromRaw[pid];
      const b = toRaw[pid];
      if (!a || !b || Math.hypot(a.x - b.x, a.y - b.y) < 8) return;
      out.push({
        kind: pid === 'o1' ? 'cut' : 'relocate',
        label: `${pid === 'o1' ? 'Cut' : 'Relocate'} ${pid.replace(/^o/, '#')}`,
        path: {
          pid,
          ballOnly: false,
          from: { x: a.x, y: a.y },
          to: { x: b.x, y: b.y },
          ctrl: movementQuadControl(a, b, 0.12),
          points: null,
          type: 'cut',
        },
        holdAfterMs: pid === 'o1' ? ANIM.HOLD_CUT_MS : Math.min(ANIM.HOLD_RELOCATE_MS, 280),
        ballTo: null,
      });
    });
  } else {
    const dropPath = (pathList || []).find((p) => p && p.pitt5Drop && p.pid === 'o4');
    if (dropPath && !out.some((b) => b.path && b.path.pitt5Drop)) {
      out.push({
        kind: 'relocate',
        label: 'Relocate #4',
        path: dropPath,
        holdAfterMs: ANIM.HOLD_RELOCATE_MS,
        ballTo: null,
      });
    }
  }
  return out;
}
